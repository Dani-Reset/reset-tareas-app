// Formatea una fecha ISO como texto relativo sencillo
function fechaRelativa(iso){
if(!iso) return '';
const d=new Date(iso), ahora=new Date();
const min=Math.floor((ahora-d)/60000);
if(min<1) return 'ahora';
if(min<60) return 'hace '+min+' min';
const h=Math.floor(min/60);
if(h<24) return 'hace '+h+' h';
const dias=Math.floor(h/24);
if(dias<7) return 'hace '+dias+' d';
return d.toLocaleDateString('es-ES',{day:'numeric',month:'short'});
}
// Convierte enlaces en texto a <a> clicables y resalta @menciones
function formatearTexto(txt){
if(!txt) return '';
let out=txt.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
out=out.replace(/(https?:\/\/[^\s]+)/g,'<a href="$1" target="_blank" class="link-inline">$1</a>');
out=out.replace(/@(Alba|Daniela)/g,'<span class="mention">@$1</span>');
return out.replace(/\n/g,'<br>');
}

function renderIdeas(){
const cont=document.getElementById('ideasList'); if(!cont) return;
if(!ideas.length){cont.innerHTML='<div style="font-size:13px; color:var(--ink-faint); padding:6px 0;">Aún no hay ideas. Añade la primera abajo.</div>';return;}
cont.innerHTML=ideas.map(i=>`<div style="display:flex; align-items:center; gap:10px; padding:8px 10px; background:var(--card); border:0.5px solid var(--line); border-radius:6px; margin-bottom:5px;">
<span style="color:var(--kiwi);">💡</span>
<span style="flex:1; font-size:14px;">${i.texto}</span>
<button class="btn btn-ghost" style="padding:3px 9px; font-size:11px;" onclick="ideaAPublicacion(${i.id})" title="Convertir en publicación programada">📅 Programar</button>
<button class="btn btn-ghost" style="padding:3px 8px; font-size:12px; color:var(--coral);" onclick="borrarIdea(${i.id})" title="Eliminar">✕</button>
</div>`).join('');
}
async function addIdea(){
const inp=document.getElementById('ideaInput');
const txt=inp.value.trim();
if(!txt){showToast('Escribe una idea primero');return;}
const {error}=await sb.from('ideas').insert({texto:txt, creado_por:miPerfil?miPerfil.id:null});
if(error){showToast('Error al guardar idea: '+error.message);return;}
inp.value='';
await cargarIdeas();
renderIdeas();
showToast('Idea añadida');
}
async function borrarIdea(id){
const {error}=await sb.from('ideas').delete().eq('id',id);
if(error){showToast('Error al eliminar: '+error.message);return;}
ideas=ideas.filter(i=>i.id!==id); renderIdeas(); showToast('Idea eliminada');
}

const idea=ideas.find(i=>i.id===id); if(!idea) return;
const texto=idea.texto;
// Borrar la idea (se transformará en publicación)

renderIdeas();
openCreateModal(null,true);
document.getElementById('fTitle').value=texto;
showToast('Convierte la idea en publicación: ponle fecha y guarda');
}
